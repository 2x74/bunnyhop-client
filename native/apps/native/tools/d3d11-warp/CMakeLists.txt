cmake_minimum_required(VERSION 3.24)
project(d3d11_warp_probe LANGUAGES CXX)
add_executable(d3d11-warp-probe probe.cpp)
target_compile_features(d3d11-warp-probe PRIVATE cxx_std_20)
target_compile_definitions(d3d11-warp-probe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(d3d11-warp-probe PRIVATE ../../vendor)
target_link_libraries(d3d11-warp-probe PRIVATE d3d11 dxgi d3dcompiler windowscodecs ole32 user32)
if(MSVC)
    target_compile_options(d3d11-warp-probe PRIVATE /W4 /EHsc)
endif()
