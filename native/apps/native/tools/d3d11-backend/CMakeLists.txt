cmake_minimum_required(VERSION 3.24)
project(d3d11_backend_proof LANGUAGES CXX RC)
include(FetchContent)
file(TO_CMAKE_PATH "${SHADER_RESOURCES}" SHADER_RESOURCES)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../packaging/sdl.json" SDL_PIN)
string(JSON SDL_COMMIT GET "${SDL_PIN}" commit)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
FetchContent_Declare(SDL3 URL "https://github.com/libsdl-org/SDL/archive/${SDL_COMMIT}.zip")
FetchContent_MakeAvailable(SDL3)
add_executable(d3d11-backend-proof ../../src/gfx_smoke.cpp ../../src/gfx_d3d11.cpp
    "${SHADER_RESOURCES}/d3d11-shaders.rc")
target_compile_features(d3d11-backend-proof PRIVATE cxx_std_20)
target_compile_definitions(d3d11-backend-proof PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(d3d11-backend-proof PRIVATE ../../src ../../include ../../vendor)
target_link_libraries(d3d11-backend-proof PRIVATE SDL3::SDL3-static d3d11 opengl32)
if(MSVC)
    target_compile_options(d3d11-backend-proof PRIVATE /W4 /EHsc /utf-8 /bigobj)
endif()
enable_testing()
add_test(NAME native_gfx_d3d11 COMMAND d3d11-backend-proof "${CMAKE_CURRENT_BINARY_DIR}/evidence" --warp)
