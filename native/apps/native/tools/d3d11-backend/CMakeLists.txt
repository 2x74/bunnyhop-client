cmake_minimum_required(VERSION 3.24)
project(d3d11_backend_proof LANGUAGES CXX RC)
include(FetchContent)
file(TO_CMAKE_PATH "${SHADER_RESOURCES}" SHADER_RESOURCES)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../packaging/sdl.json" SDL_PIN)
string(JSON SDL_COMMIT GET "${SDL_PIN}" commit)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
FetchContent_Declare(SDL3 URL "https://github.com/libsdl-org/SDL/archive/${SDL_COMMIT}.zip")
FetchContent_MakeAvailable(SDL3)
function(backend_proof target source)
    add_executable(${target} "../../src/${source}" ../../src/gfx_d3d11.cpp
        "${SHADER_RESOURCES}/d3d11-shaders.rc")
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_include_directories(${target} PRIVATE ../../src ../../include ../../vendor)
    target_link_libraries(${target} PRIVATE SDL3::SDL3-static d3d11 opengl32)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /EHsc /utf-8 /bigobj)
    endif()
endfunction()
backend_proof(d3d11-backend-proof gfx_smoke.cpp)
backend_proof(d3d11-post-proof post_processing_smoke.cpp)
enable_testing()
add_test(NAME native_gfx_d3d11 COMMAND d3d11-backend-proof "${CMAKE_CURRENT_BINARY_DIR}/evidence" --warp)
add_test(NAME native_gfx_opengl COMMAND d3d11-backend-proof)
set_tests_properties(native_gfx_opengl PROPERTIES SKIP_RETURN_CODE 77)
add_test(NAME native_post_processing_d3d11 COMMAND d3d11-post-proof --warp)
