# Keep diagnostics on native code, including Objective-C++ platform adapters.
if(MSVC)
    # Sources and string literals are UTF-8 (HUD text uses "·" and "…").
    add_compile_options(/W4 /utf-8)
elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    add_compile_options(-Wall -Wextra)
endif()
if(WIN32)
    # Windows headers pulled in by SDL/cURL must not replace C++ min/max calls.
    add_compile_definitions(NOMINMAX)

# package.ts refuses to make a release package from a binary built this way.
option(BUNNYHOP_UPDATE_TEST_HOOKS "Build the update end-to-end test hooks into the client" OFF)

# Keep release executables free of local checkout / Cargo-cache paths. Preserve
# caller flags, using Cargo's separator form so paths containing spaces work.
string(ASCII 31 BUNNYHOP_FLAG_SEPARATOR)
if(DEFINED ENV{CARGO_ENCODED_RUSTFLAGS})
    set(BUNNYHOP_RUST_FLAGS "$ENV{CARGO_ENCODED_RUSTFLAGS}")
else()
    separate_arguments(BUNNYHOP_CALLER_RUST_FLAGS NATIVE_COMMAND "$ENV{RUSTFLAGS}")
    list(JOIN BUNNYHOP_CALLER_RUST_FLAGS "${BUNNYHOP_FLAG_SEPARATOR}" BUNNYHOP_RUST_FLAGS)
endif()

set(BUNNYHOP_REMAPS)

    list(APPEND BUNNYHOP_REMAPS "${BUNNYHOP_USER_HOME}=/build-user")
endif()
if(BUNNYHOP_CARGO_HOME)
    list(APPEND BUNNYHOP_REMAPS "${BUNNYHOP_CARGO_HOME}=/rust-cargo")
endif()
list(APPEND BUNNYHOP_REMAPS "${BUNNYHOP_ROOT}=/bunnyhop-native")
foreach(BUNNYHOP_REMAP IN LISTS BUNNYHOP_REMAPS)
    if(BUNNYHOP_RUST_FLAGS)
        string(APPEND BUNNYHOP_RUST_FLAGS "${BUNNYHOP_FLAG_SEPARATOR}")
    endif()
    string(APPEND BUNNYHOP_RUST_FLAGS "--remap-path-prefix=${BUNNYHOP_REMAP}")
    if(MSVC)
        add_compile_options("/pathmap:${BUNNYHOP_REMAP}")
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        add_compile_options("-ffile-prefix-map=${BUNNYHOP_REMAP}")
    endif()
endforeach()
set(RUST_TARGET_DIR "${BUNNYHOP_ROOT}/target" CACHE PATH "Cargo build output directory")
set(RUST_LIBRARY "${RUST_TARGET_DIR}/release/${CMAKE_STATIC_LIBRARY_PREFIX}bunnyhop_native_bridge${CMAKE_STATIC_LIBRARY_SUFFIX}")
file(GLOB_RECURSE RUST_SOURCES CONFIGURE_DEPENDS
    "${BUNNYHOP_ROOT}/crates/*.rs" "${BUNNYHOP_ROOT}/crates/*/Cargo.toml")
add_custom_command(OUTPUT "${RUST_LIBRARY}"
    COMMAND "${CMAKE_COMMAND}" -E env "CARGO_TARGET_DIR=${RUST_TARGET_DIR}"
        "CARGO_ENCODED_RUSTFLAGS=${BUNNYHOP_RUST_FLAGS}"
        "${CARGO_EXECUTABLE}" build --manifest-path "${BUNNYHOP_ROOT}/Cargo.toml"
        --release --locked -p bunnyhop-native-bridge --lib
    DEPENDS ${RUST_SOURCES} "${BUNNYHOP_ROOT}/Cargo.toml" "${BUNNYHOP_ROOT}/Cargo.lock"
    WORKING_DIRECTORY "${BUNNYHOP_ROOT}"
    COMMENT "Building the shared bunnyhop Rust simulation and native ABI"
    VERBATIM)

if(BUILD_TESTING)
    add_executable(bunnyhop-native-longjump-tests src/longjump_tests.cpp)
    target_include_directories(bunnyhop-native-longjump-tests PRIVATE include vendor src)
    target_compile_options(bunnyhop-native-longjump-tests PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/UNDEBUG> $<$<NOT:$<CXX_COMPILER_ID:MSVC>>:-UNDEBUG>)
    find_program(LONGJUMP_BUN bun)
    if(LONGJUMP_BUN)
        set(LONGJUMP_ORACLE "${CMAKE_CURRENT_BINARY_DIR}/longjump-oracle.json")
        add_test(NAME native_longjump_oracle COMMAND "${LONGJUMP_BUN}" apps/native/src/longjump_oracle.ts "${LONGJUMP_ORACLE}")
        set_tests_properties(native_longjump_oracle PROPERTIES WORKING_DIRECTORY "${BUNNYHOP_ROOT}" FIXTURES_SETUP longjump_oracle)
        add_test(NAME native_longjump COMMAND bunnyhop-native-longjump-tests "${LONGJUMP_ORACLE}")
        set_tests_properties(native_longjump PROPERTIES FIXTURES_REQUIRED longjump_oracle)
    endif()
endif()
