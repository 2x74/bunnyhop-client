cmake_minimum_required(VERSION 3.24)
project(bunnyhop_native VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BUNNYHOP_BUILD_CLIENT "Build the SDL3/OpenGL desktop client" ON)
option(BUNNYHOP_TEST_GTK "Run GTK lifecycle tests under Xvfb and D-Bus" OFF)
get_filename_component(BUNNYHOP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_program(CARGO_EXECUTABLE cargo REQUIRED)
# Keep release executables free of local checkout / Cargo-cache paths. Preserve
# caller flags, using Cargo's separator form so paths containing spaces work.
string(ASCII 31 BUNNYHOP_FLAG_SEPARATOR)
if(DEFINED ENV{CARGO_ENCODED_RUSTFLAGS})
    set(BUNNYHOP_RUST_FLAGS "$ENV{CARGO_ENCODED_RUSTFLAGS}")
else()
    separate_arguments(BUNNYHOP_CALLER_RUST_FLAGS NATIVE_COMMAND "$ENV{RUSTFLAGS}")
    list(JOIN BUNNYHOP_CALLER_RUST_FLAGS "${BUNNYHOP_FLAG_SEPARATOR}" BUNNYHOP_RUST_FLAGS)
endif()
set(BUNNYHOP_USER_HOME "$ENV{HOME}")
if(NOT BUNNYHOP_USER_HOME)
    set(BUNNYHOP_USER_HOME "$ENV{USERPROFILE}")
endif()
set(BUNNYHOP_CARGO_HOME "$ENV{CARGO_HOME}")
if(NOT BUNNYHOP_CARGO_HOME AND BUNNYHOP_USER_HOME)
    set(BUNNYHOP_CARGO_HOME "${BUNNYHOP_USER_HOME}/.cargo")
endif()
set(BUNNYHOP_REMAPS)
if(BUNNYHOP_USER_HOME)
    list(APPEND BUNNYHOP_REMAPS "${BUNNYHOP_USER_HOME}=/build-user")
endif()
if(BUNNYHOP_CARGO_HOME)
    list(APPEND BUNNYHOP_REMAPS "${BUNNYHOP_CARGO_HOME}=/rust-cargo")
endif()
list(APPEND BUNNYHOP_REMAPS "${BUNNYHOP_ROOT}=/bunnyhop-native")
foreach(BUNNYHOP_REMAP IN LISTS BUNNYHOP_REMAPS)
    if(BUNNYHOP_RUST_FLAGS)
        string(APPEND BUNNYHOP_RUST_FLAGS "${BUNNYHOP_FLAG_SEPARATOR}")
    endif()
    string(APPEND BUNNYHOP_RUST_FLAGS "--remap-path-prefix=${BUNNYHOP_REMAP}")
    if(MSVC)
        add_compile_options("/pathmap:${BUNNYHOP_REMAP}")
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        add_compile_options("-ffile-prefix-map=${BUNNYHOP_REMAP}")
    endif()
endforeach()
set(RUST_TARGET_DIR "${BUNNYHOP_ROOT}/target" CACHE PATH "Cargo build output directory")
set(RUST_LIBRARY "${RUST_TARGET_DIR}/release/${CMAKE_STATIC_LIBRARY_PREFIX}bunnyhop_native_bridge${CMAKE_STATIC_LIBRARY_SUFFIX}")
file(GLOB_RECURSE RUST_SOURCES CONFIGURE_DEPENDS
    "${BUNNYHOP_ROOT}/crates/*.rs" "${BUNNYHOP_ROOT}/crates/*/Cargo.toml")
add_custom_command(OUTPUT "${RUST_LIBRARY}"
    COMMAND "${CMAKE_COMMAND}" -E env "CARGO_TARGET_DIR=${RUST_TARGET_DIR}"
        "CARGO_ENCODED_RUSTFLAGS=${BUNNYHOP_RUST_FLAGS}"
        "${CARGO_EXECUTABLE}" build --manifest-path "${BUNNYHOP_ROOT}/Cargo.toml"
        --release --locked -p bunnyhop-native-bridge --lib
    DEPENDS ${RUST_SOURCES} "${BUNNYHOP_ROOT}/Cargo.toml" "${BUNNYHOP_ROOT}/Cargo.lock"
    WORKING_DIRECTORY "${BUNNYHOP_ROOT}"
    COMMENT "Building the shared bunnyhop Rust simulation and native ABI"
    VERBATIM)
add_custom_target(bunnyhop_rust DEPENDS "${RUST_LIBRARY}")
add_library(bunnyhop_core_native STATIC IMPORTED GLOBAL)
set_target_properties(bunnyhop_core_native PROPERTIES IMPORTED_LOCATION "${RUST_LIBRARY}")
add_dependencies(bunnyhop_core_native bunnyhop_rust)
target_include_directories(bunnyhop_core_native INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
if(APPLE)
    target_link_libraries(bunnyhop_core_native INTERFACE "-framework Security" "-framework CoreFoundation" iconv resolv)
elseif(UNIX)
    find_package(Threads REQUIRED)
    target_link_libraries(bunnyhop_core_native INTERFACE Threads::Threads ${CMAKE_DL_LIBS} m)
elseif(WIN32)
    target_link_libraries(bunnyhop_core_native INTERFACE ws2_32 userenv bcrypt ntdll)
endif()
add_executable(bunnyhop-native-runner src/runner.cpp)
target_link_libraries(bunnyhop-native-runner PRIVATE bunnyhop_core_native)
include(CTest)
if(BUNNYHOP_BUILD_CLIENT)
    find_package(SDL3 CONFIG REQUIRED)
    find_package(OpenGL REQUIRED)
    find_package(CURL REQUIRED)
    find_package(ZLIB REQUIRED)
    find_package(Threads REQUIRED)
    add_executable(bunnyhop-native src/client.cpp src/online.cpp src/audio_pack.cpp)
    target_include_directories(bunnyhop-native PRIVATE vendor)
    target_link_libraries(bunnyhop-native PRIVATE bunnyhop_core_native SDL3::SDL3 OpenGL::GL CURL::libcurl Threads::Threads ZLIB::ZLIB)
    if(APPLE)
        enable_language(OBJCXX)
        target_sources(bunnyhop-native PRIVATE src/menu_webview_macos.mm)
        set_source_files_properties(src/menu_webview_macos.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
        target_link_libraries(bunnyhop-native PRIVATE "-framework Cocoa" "-framework WebKit")
    elseif(WIN32)
        include(packaging/WebView2.cmake)
        bunnyhop_windows_menu(bunnyhop-native)
    elseif(UNIX)
        find_package(PkgConfig QUIET)
        if(PkgConfig_FOUND)
            pkg_check_modules(WEBKITGTK QUIET IMPORTED_TARGET webkit2gtk-4.1>=2.40 gtk+-3.0)
        endif()
        if(WEBKITGTK_FOUND)
            target_sources(bunnyhop-native PRIVATE src/menu_webview_gtk.cpp)
            target_link_libraries(bunnyhop-native PRIVATE PkgConfig::WEBKITGTK)
            if(BUILD_TESTING)
                add_executable(bunnyhop-native-menu-gl-tests src/menu_gl_context_tests.cpp)
                target_include_directories(bunnyhop-native-menu-gl-tests PRIVATE
                    $<TARGET_PROPERTY:SDL3::SDL3,INTERFACE_INCLUDE_DIRECTORIES>)
                add_test(NAME native_menu_gl_boundaries COMMAND bunnyhop-native-menu-gl-tests)
                add_executable(bunnyhop-native-menu-gtk-tests src/menu_webview_gtk_tests.cpp src/menu_webview_gtk.cpp)
                target_link_libraries(bunnyhop-native-menu-gtk-tests PRIVATE SDL3::SDL3 OpenGL::GL PkgConfig::WEBKITGTK)
                if(BUNNYHOP_TEST_GTK)
                    find_program(XVFB_RUN xvfb-run REQUIRED)
                    find_program(DBUS_RUN_SESSION dbus-run-session REQUIRED)
                    add_test(NAME native_menu_gtk_lifecycle COMMAND "${CMAKE_COMMAND}" -E env
                        SDL_VIDEO_DRIVER=x11 LIBGL_ALWAYS_SOFTWARE=1
                        "${XVFB_RUN}" -a "${DBUS_RUN_SESSION}" --
                        $<TARGET_FILE:bunnyhop-native-menu-gtk-tests> "${CMAKE_CURRENT_BINARY_DIR}/gtk-boundary.png")
                    set_tests_properties(native_menu_gtk_lifecycle PROPERTIES TIMEOUT 60)
                endif()
            endif()
        else()
            message(STATUS "WebKitGTK 4.1 not installed: native game available with --map/--online-map; menus disabled")
            target_sources(bunnyhop-native PRIVATE src/menu_webview_stub.cpp)
        endif()
    else()
        target_sources(bunnyhop-native PRIVATE src/menu_webview_stub.cpp)
    endif()
    if(BUILD_TESTING)
        add_executable(bunnyhop-native-client-tests src/client_boundary_tests.cpp src/audio_pack.cpp)
        target_include_directories(bunnyhop-native-client-tests PRIVATE vendor include src)
        target_link_libraries(bunnyhop-native-client-tests PRIVATE SDL3::SDL3)
        add_test(NAME native_client_boundaries COMMAND bunnyhop-native-client-tests)
        add_executable(bunnyhop-native-online-tests src/online_tests.cpp src/online.cpp)
        target_include_directories(bunnyhop-native-online-tests PRIVATE vendor)
        target_link_libraries(bunnyhop-native-online-tests PRIVATE bunnyhop_core_native CURL::libcurl)
        add_test(NAME native_online_boundaries COMMAND bunnyhop-native-online-tests)
    endif()
    target_compile_definitions(bunnyhop-native PRIVATE
        BUNNYHOP_MENU_HTML="target/native-menu/menu.html"
        BUNNYHOP_NATIVE_MANIFEST="target/native-menu/native-manifest.json"
        BUNNYHOP_NATIVE_ASSET_ROOT="target/native-assets")
    # Explicit framebuffer check: run on a desktop or under Xvfb/Mesa, not in
    # ordinary headless CTest. Does not initialize a map or gameplay context.
    add_executable(bunnyhop-native-menu-scene-smoke src/menu_scene_smoke.cpp)
    target_include_directories(bunnyhop-native-menu-scene-smoke PRIVATE vendor include)
    target_link_libraries(bunnyhop-native-menu-scene-smoke PRIVATE SDL3::SDL3 OpenGL::GL ZLIB::ZLIB)
    add_executable(bunnyhop-native-render-smoke src/render_smoke.cpp)
    target_include_directories(bunnyhop-native-render-smoke PRIVATE vendor)
    target_link_libraries(bunnyhop-native-render-smoke PRIVATE bunnyhop_core_native SDL3::SDL3 OpenGL::GL ZLIB::ZLIB)
    target_compile_definitions(bunnyhop-native-render-smoke PRIVATE
        BUNNYHOP_NATIVE_ASSET_ROOT="target/native-assets")
    find_program(BUN_EXECUTABLE bun)
    if(BUN_EXECUTABLE)
        add_custom_target(bunnyhop-native-menu
            COMMAND "${BUN_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/build-menu.ts"
            WORKING_DIRECTORY "${BUNNYHOP_ROOT}"
            COMMENT "Building menus and account UI (no web game renderer)")
    endif()
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(bunnyhop-native PRIVATE -ffp-contract=off)
        target_compile_options(bunnyhop-native-render-smoke PRIVATE -ffp-contract=off)
    endif()
endif()
if(BUILD_TESTING)
    add_executable(bunnyhop-native-room-modes-tests src/room_modes_tests.cpp)
    target_include_directories(bunnyhop-native-room-modes-tests PRIVATE vendor)
    add_test(NAME native_room_modes COMMAND bunnyhop-native-room-modes-tests)
    add_executable(bunnyhop-native-tutorial-tests src/tutorial_tests.cpp)
    target_include_directories(bunnyhop-native-tutorial-tests PRIVATE include vendor)
    find_program(BUN_EXECUTABLE bun)
    if(BUN_EXECUTABLE)
        set(TUTORIAL_ORACLE "${CMAKE_CURRENT_BINARY_DIR}/tutorial-oracle.json")
        add_test(NAME native_tutorial_oracle COMMAND "${BUN_EXECUTABLE}" apps/native/src/tutorial_oracle.ts "${TUTORIAL_ORACLE}")
        set_tests_properties(native_tutorial_oracle PROPERTIES
            WORKING_DIRECTORY "${BUNNYHOP_ROOT}" FIXTURES_SETUP tutorial_oracle TIMEOUT 120)
        add_test(NAME native_tutorial COMMAND bunnyhop-native-tutorial-tests "${TUTORIAL_ORACLE}")
        set_tests_properties(native_tutorial PROPERTIES FIXTURES_REQUIRED tutorial_oracle)
        # The jump-stats port is compared against the web reference the same way: the oracle runs
        # the TypeScript over a fixed tick stream, the test replays it through the C++.
        set(SSJ_ORACLE "${CMAKE_CURRENT_BINARY_DIR}/ssj-oracle.json")
        add_test(NAME native_ssj_oracle COMMAND "${BUN_EXECUTABLE}" apps/native/src/ssj_oracle.ts "${SSJ_ORACLE}")
        set_tests_properties(native_ssj_oracle PROPERTIES
            WORKING_DIRECTORY "${BUNNYHOP_ROOT}" FIXTURES_SETUP ssj_oracle TIMEOUT 120)
        add_test(NAME native_ssj COMMAND bunnyhop-native-ssj-tests "${SSJ_ORACLE}")
        set_tests_properties(native_ssj PROPERTIES FIXTURES_REQUIRED ssj_oracle)
    endif()
    add_executable(bunnyhop-native-ssj-tests src/ssj_tests.cpp)
    target_include_directories(bunnyhop-native-ssj-tests PRIVATE vendor src)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        # The port reproduces SourcePawn's 32-bit arithmetic; contracting a multiply and an add
        # into an FMA would quietly compute it at a different width.
        target_compile_options(bunnyhop-native-ssj-tests PRIVATE -ffp-contract=off)
    endif()
    add_test(NAME native_runner_help COMMAND bunnyhop-native-runner --help)
    # Decoding and pack scanning need no SDL, so these run everywhere.
    add_executable(bunnyhop-native-audio-tests src/audio_pack_tests.cpp src/audio_pack.cpp)
    target_include_directories(bunnyhop-native-audio-tests PRIVATE vendor src)
    add_test(NAME native_audio_pack COMMAND bunnyhop-native-audio-tests)
    add_executable(bunnyhop-native-input-tests src/native_input_tests.cpp)
    target_include_directories(bunnyhop-native-input-tests PRIVATE include)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(bunnyhop-native-input-tests PRIVATE -ffp-contract=off)
    endif()
    add_test(NAME native_input COMMAND bunnyhop-native-input-tests)
    # Discord RPC: framing, Join secrets, privacy rules, and a fake Discord socket.
    add_executable(bunnyhop-native-discord-tests src/discord_rpc_tests.cpp)
    target_include_directories(bunnyhop-native-discord-tests PRIVATE vendor src)
    add_test(NAME native_discord COMMAND bunnyhop-native-discord-tests)
endif()
